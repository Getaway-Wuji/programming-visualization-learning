for(const spec of specs)test(`${spec.kind} actual zero pending publication, residual and scope removal restore exactly`,async({page:p})=>{
  await observeCaseSetExecution(p);await ready(p,spec);await run(p,'入 0\n出');const {result}=await captured(p,`${spec.kind}-pending`);
  const record=result.trace!.steps.flatMap(s=>s.changes).find(c=>c.op==='record'&&c.table==='objects'&&c.after&&'name'in c.after&&c.after.name==='data')!;
  if(record.op!=='record')throw Error('Actual data declaration required');
  const append=result.trace!.steps.find(s=>s.changes.some(c=>c.op==='cell'&&c.objectId===record.id&&c.index===0&&c.after===0))!;
  await seek(p,append.seq);await sameState(p,result,append.seq,spec);
  const cell=p.locator('[data-testid="structure-cell"][data-index="0"]');await expect(cell).toHaveAttribute('data-value','0');await expect(cell).toHaveAttribute('data-valid','false');await expect(cell).toHaveAttribute('data-write','true');
  await seek(p,append.seq-1);await sameState(p,result,append.seq-1,spec);await expect(cell).toHaveAttribute('data-value','uninitialized');
  const player=new AlgorithmPlayer();player.accept(result);let emptyAfter=0;for(let i=1;i<=player.snapshot().length;i++){player.step(1);if(scalarValue(player.snapshot().state!,'processed')===2){emptyAfter=i;break;}}
  await seek(p,emptyAfter);await sameState(p,result,emptyAfter,spec);await expect(cell).toHaveAttribute('data-value','0');await expect(cell).toHaveAttribute('data-valid','false');
  const removal=result.trace!.steps.find(s=>s.changes.some(c=>c.op==='record'&&c.table==='objects'&&c.id===record.id&&!c.after))!;
  await seek(p,removal.seq);await sameState(p,result,removal.seq,spec);await expect(p.getByTestId('structure-cell')).toHaveCount(0);
  await seek(p,removal.seq-1);await sameState(p,result,removal.seq-1,spec);await expect(p.getByTestId('structure-cell')).toHaveCount(5);
});
